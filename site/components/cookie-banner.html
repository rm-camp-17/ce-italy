'granted' : 'denied'; // Uncomment the following block when GA4 / GTM is installed: /* if (typeof gtag === 'function') { gtag('consent', 'update', { 'analytics_storage': status, 'ad_storage': status, 'ad_user_data': status, 'ad_personalization': status }); } */ // Optional: dispatch a custom event for other scripts to listen to try { window.dispatchEvent(new CustomEvent('cookieConsentUpdate', { detail: { consent: granted } })); } catch (e) { // Fallback for older browsers } } /** * Save user preference and hide the banner. * @param {string} choice - 'accepted' or 'rejected' */ function saveChoice(choice) { try { localStorage.setItem(STORAGE_KEY, JSON.stringify({ choice: choice, timestamp: new Date().toISOString() })); } catch (e) { // localStorage not available (private browsing, etc.) } updateConsent(choice === 'accepted'); hideBanner(); } /** * Show the cookie banner with slide-up animation. */ function showBanner() { banner.style.display = 'block'; // Force a reflow before adding the animation class banner.offsetHeight; banner.classList.add('is-visible'); // Focus the accept button for accessibility acceptBtn.focus(); } /** * Hide the cookie banner with slide-down animation. */ function hideBanner() { banner.classList.remove('is-visible'); banner.classList.add('is-hiding'); // Wait for animation to complete before hiding setTimeout(function() { banner.style.display = 'none'; banner.classList.remove('is-hiding'); }, 400); } /** * Check if user has already made a choice. * @returns {object|null} The stored consent data, or null. */ function getStoredConsent() { try { var stored = localStorage.getItem(STORAGE_KEY); return stored ? JSON.parse(stored) : null; } catch (e) { return null; } } // Event listeners acceptBtn.addEventListener('click', function() { saveChoice('accepted'); }); rejectBtn.addEventListener('click', function() { saveChoice('rejected'); }); // Initialize: only show banner if no consent has been recorded var consent = getStoredConsent(); if (consent) { // User already made a choice - apply it silently updateConsent(consent.choice === 'accepted'); } else { // No choice recorded - show the banner after a short delay setTimeout(showBanner, 800); } })();
